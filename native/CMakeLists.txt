cmake_minimum_required(VERSION 3.20)
project(gemmajev LANGUAGES C CXX)

if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
    message(FATAL_ERROR "GemmaJev currently supports macOS on Apple Silicon")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GEMMAJEV_VISION "Build the optional image-scoring worker" OFF)
set(GEMMAJEV_LLAMA_SOURCE "" CACHE PATH "Private pinned llama.cpp source")
set(GEMMAJEV_LLAMA_REVISION "8086439a4cea94c71a5dfb8fe4ad1546aebd640f")
set(revision_file "${GEMMAJEV_LLAMA_SOURCE}/.gemmajev-revision")
if(NOT EXISTS "${revision_file}")
    message(FATAL_ERROR "Prepare the private source with scripts/build_runtime.sh")
endif()
file(READ "${revision_file}" revision)
string(STRIP "${revision}" revision)
if(NOT revision STREQUAL GEMMAJEV_LLAMA_REVISION)
    message(FATAL_ERROR "Source revision differs from the pinned llama.cpp revision")
endif()
set(patch_file "${CMAKE_CURRENT_SOURCE_DIR}/patches/gemma4-raw-logits.patch")
set(patch_stamp "${GEMMAJEV_LLAMA_SOURCE}/.gemmajev-raw-logits.sha256")
if(NOT EXISTS "${patch_stamp}")
    message(FATAL_ERROR "Apply the raw-logits patch with scripts/build_runtime.sh")
endif()
file(SHA256 "${patch_file}" patch_hash)
file(READ "${patch_stamp}" applied_hash)
string(STRIP "${applied_hash}" applied_hash)
if(NOT patch_hash STREQUAL applied_hash)
    message(FATAL_ERROR "Runtime patch changed; run scripts/build_runtime.sh")
endif()
execute_process(COMMAND patch -p1 -f -R --dry-run -i "${patch_file}"
    WORKING_DIRECTORY "${GEMMAJEV_LLAMA_SOURCE}" RESULT_VARIABLE patch_status
    OUTPUT_QUIET ERROR_QUIET)
if(NOT patch_status EQUAL 0)
    message(FATAL_ERROR "The private source does not contain the recorded raw-logits patch")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/gemmajev.cpp" "${revision_file}" "${patch_file}" "${patch_stamp}")

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(feature TESTS EXAMPLES TOOLS SERVER APP COMMON)
    set(LLAMA_BUILD_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
if(GEMMAJEV_VISION)
    set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
    set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
    set(image_patch "${CMAKE_CURRENT_SOURCE_DIR}/patches/gemma4-image-budget.patch")
    file(SHA256 "${image_patch}" image_patch_hash)
    execute_process(COMMAND patch -p1 -f -R --dry-run -i "${image_patch}"
        WORKING_DIRECTORY "${GEMMAJEV_LLAMA_SOURCE}" RESULT_VARIABLE image_patch_status
        OUTPUT_QUIET ERROR_QUIET)
    if(NOT image_patch_status EQUAL 0)
        message(FATAL_ERROR "Apply the image budget patch using GEMMAJEV_VISION=1 scripts/build_runtime.sh")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${image_patch}")
endif()
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
add_subdirectory("${GEMMAJEV_LLAMA_SOURCE}" llama EXCLUDE_FROM_ALL)
if(GEMMAJEV_VISION)
    get_directory_property(LLAMA_INSTALL_VERSION DIRECTORY "${GEMMAJEV_LLAMA_SOURCE}" DEFINITION LLAMA_INSTALL_VERSION)
    add_subdirectory("${GEMMAJEV_LLAMA_SOURCE}/tools/mtmd" mtmd EXCLUDE_FROM_ALL)
endif()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/gemmajev.cpp" worker_hash)
add_executable(gemmajev-worker gemmajev.cpp)
set_target_properties(gemmajev-worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/staging")
target_include_directories(gemmajev-worker PRIVATE "${GEMMAJEV_LLAMA_SOURCE}/vendor")
target_link_libraries(gemmajev-worker PRIVATE llama)
target_compile_definitions(gemmajev-worker PRIVATE
    GEMMAJEV_LLAMA_REVISION="${GEMMAJEV_LLAMA_REVISION}"
    GEMMAJEV_PATCH_SHA256="${patch_hash}"
    GEMMAJEV_WORKER_SHA256="${worker_hash}")
target_compile_options(gemmajev-worker PRIVATE -Wall -Wextra -Wpedantic)
if(GEMMAJEV_VISION)
    set_target_properties(gemmajev-worker PROPERTIES OUTPUT_NAME gemmajev-vision-worker)
    target_link_libraries(gemmajev-worker PRIVATE mtmd)
    target_compile_definitions(gemmajev-worker PRIVATE
        GEMMAJEV_VISION=1 GEMMAJEV_IMAGE_PATCH_SHA256="${image_patch_hash}")
endif()
